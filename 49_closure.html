// ========================================== // 📘 CLOSURES IN JAVASCRIPT // ========================================== // Closure = Function + // Access to its outer scope variables // Even after outer function finishes, // inner function remembers variables // ========================================== // 📘 Basic Closure Example // ========================================== // function init() { // let name = "Mozilla"; // function displayName() { // console.log(name); // } // displayName(); // } // init(); // Output: // Mozilla // displayName() can access // variable from outer function // ========================================== // 📘 Lexical Scoping // ========================================== // Inner functions can access // variables of outer functions // function outer(){ // let username = "hitesh" // function inner(){ // let secret = "my123" // console.log( // "inner", // username // ); // } // function innerTwo(){ // console.log( // "innerTwo", // username // ); // // ❌ Cannot access // // secret from inner() // console.log(secret); // } // inner() // innerTwo() // } // outer(); // Rules: // inner() can access outer() // outer() cannot access inner() // ========================================== // 📘 Real Closure Example // ========================================== // function makeFunc(){ // const name = "Mozilla"; // function displayName(){ // console.log(name); // } // return displayName; // } // const myFunc = makeFunc() // myFunc() // Output: // Mozilla // Even though makeFunc() // finished execution, // displayName still remembers // variable name // This is Closure // ========================================== // 📘 Without Closure // ========================================== // document.getElementById("orange") // .onclick = function(){ // document.body.style // .backgroundColor = "orange" // } // document.getElementById("green") // .onclick = function(){ // document.body.style // .backgroundColor = "green" // } // Repetitive code // ========================================== // 📘 Closure Example with Buttons // ========================================== function clickHandler(color){ // color becomes part // of closure return function(){ document.body.style .backgroundColor = color } } // Create closure for orange document.getElementById('orange') .onclick = clickHandler("orange") // Create closure for green document.getElementById('green') .onclick = clickHandler("green") // ========================================== // 📘 What Happens Internally // ========================================== /* clickHandler("orange") ↓ returns function ↓ function remembers: color = "orange" ↓ button click ↓ background becomes orange -------------------------------- clickHandler("green") ↓ returns function ↓ function remembers: color = "green" ↓ button click ↓ background becomes green */ // ========================================== // 📘 Why Closure Works Here // ========================================== // Returned function remembers // color variable even after // clickHandler() finishes // This remembered variable // is called Closure // ========================================== // 📘 Interview Definition // ========================================== // Closure is a function that // remembers variables from its // lexical scope even after the // outer function has finished execution.
